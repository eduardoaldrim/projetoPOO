package modelo;

// Interface pra tudo que dá pra estimar quanto de caloria gasta.
// Quem implementa: Exercicio (e as 3 filhas, por herança) e o Treino.
// O Treino usa isso pra somar as calorias sem precisar saber o tipo de cada exercicio.
public interface Monitoravel {

    // devolve a estimativa de calorias gastas (em kcal)
    double estimarCaloriasGastas();
}

package modelo;

// Classe base de todos os exercicios. É abstrata, entao ninguem faz "new Exercicio()",
// tem que criar um de força, cardio ou funcional.
// Ela implementa Monitoravel mas deixa o calculo das calorias pra cada filha fazer do seu jeito.
public abstract class Exercicio implements Monitoravel {
    private int id;
    private String nome;
    private String grupoMuscular;
    private String descricao;

    // protected pq só as classes filhas chamam esse construtor (pelo super)
    protected Exercicio(int id, String nome, String grupoMuscular, String descricao) {
        this.id = id;
        this.nome = nome;
        this.grupoMuscular = grupoMuscular;
        this.descricao = descricao;
    }

    // cada tipo de exercicio é obrigado a implementar a sua conta
    @Override
    public abstract double estimarCaloriasGastas();

    public int getId() {
        return id;
    }

    public void setId(int id) {
        this.id = id;
    }

    public String getNome() {
        return nome;
    }

    public void setNome(String nome) {
        this.nome = nome;
    }

    public String getGrupoMuscular() {
        return grupoMuscular;
    }

    public void setGrupoMuscular(String grupoMuscular) {
        this.grupoMuscular = grupoMuscular;
    }

    public String getDescricao() {
        return descricao;
    }

    public void setDescricao(String descricao) {
        this.descricao = descricao;
    }

    // cabeçalho da tabela de exercicios, fica aqui junto do toString pra coluna bater certinho
    public static String cabecalhoTabela() {
        return String.format("%-4s | %-34s | %-14s | %-18s | %s",
                "ID", "Nome", "Grupo muscular", "Tipo", "Calorias (kcal)");
    }

    // uma linha da tabela. O estimarCaloriasGastas() aqui chama o metodo da classe filha (polimorfismo)
    @Override
    public String toString() {
        return String.format("%-4d | %-34s | %-14s | %-18s | %.1f",
                id, nome, grupoMuscular, getClass().getSimpleName(), estimarCaloriasGastas());
    }
}

package modelo;

// Exercicio de musculação, com carga (supino, agachamento, leg press...)
public class ExercicioDeForca extends Exercicio {
    private double cargaKg;
    private int series;
    private int repeticoes;
    private String equipamento;

    // os 4 primeiros vão pro construtor da classe mae, o resto é só de força
    public ExercicioDeForca(int id, String nome, String grupoMuscular, String descricao,
                            double cargaKg, int series, int repeticoes, String equipamento) {
        super(id, nome, grupoMuscular, descricao);
        this.cargaKg = cargaKg;
        this.series = series;
        this.repeticoes = repeticoes;
        this.equipamento = equipamento;
    }

    // conta simplificada: series x repetições x carga x um fator fixo (0.05)
    // não é cientifico, é só pra ter uma estimativa didatica
    @Override
    public double estimarCaloriasGastas() {
        return series * repeticoes * cargaKg * 0.05;
    }

    public double getCargaKg() {
        return cargaKg;
    }

    public void setCargaKg(double cargaKg) {
        this.cargaKg = cargaKg;
    }

    public int getSeries() {
        return series;
    }

    public void setSeries(int series) {
        this.series = series;
    }

    public int getRepeticoes() {
        return repeticoes;
    }

    public void setRepeticoes(int repeticoes) {
        this.repeticoes = repeticoes;
    }

    public String getEquipamento() {
        return equipamento;
    }

    public void setEquipamento(String equipamento) {
        this.equipamento = equipamento;
    }
}

package modelo;

import modelo.enumeradas.Intensidade;

// Exercicio aerobico: esteira, bike, eliptico, corda...
public class ExercicioCardio extends Exercicio {
    private int duracaoMinutos;
    private Intensidade intensidade;

    public ExercicioCardio(int id, String nome, String grupoMuscular, String descricao,
                           int duracaoMinutos, Intensidade intensidade) {
        super(id, nome, grupoMuscular, descricao);
        this.duracaoMinutos = duracaoMinutos;
        this.intensidade = intensidade;
    }

    // formula do MET: MET x peso x horas
    // como o exercicio não sabe quem vai fazer, usamos um peso de referencia de 70kg
    @Override
    public double estimarCaloriasGastas() {
        // cada intensidade tem um MET diferente
        double met = switch (intensidade) {
            case BAIXA -> 4.0;
            case MEDIA -> 7.0;
            case ALTA -> 10.0;
        };
        return met * 70 * (duracaoMinutos / 60.0); // divide por 60 pra passar minutos pra horas
    }

    public int getDuracaoMinutos() {
        return duracaoMinutos;
    }

    public void setDuracaoMinutos(int duracaoMinutos) {
        this.duracaoMinutos = duracaoMinutos;
    }

    public Intensidade getIntensidade() {
        return intensidade;
    }

    public void setIntensidade(Intensidade intensidade) {
        this.intensidade = intensidade;
    }
}

// Exercicio funcional / circuito: burpee, kettlebell, box jump...
public class ExercicioFuncional extends Exercicio {
    private int rodadas;
    private int tempoPorRodadaSegundos;

    public ExercicioFuncional(int id, String nome, String grupoMuscular, String descricao,
                              int rodadas, int tempoPorRodadaSegundos) {
        super(id, nome, grupoMuscular, descricao);
        this.rodadas = rodadas;
        this.tempoPorRodadaSegundos = tempoPorRodadaSegundos;
    }

    // calorias proporcionais ao tempo total: rodadas x minutos de cada rodada x 10 kcal por minuto
    @Override
    public double estimarCaloriasGastas() {
        return rodadas * (tempoPorRodadaSegundos / 60.0) * 10;
    }

    public int getRodadas() {
        return rodadas;
    }

    public void setRodadas(int rodadas) {
        this.rodadas = rodadas;
    }

    public int getTempoPorRodadaSegundos() {
        return tempoPorRodadaSegundos;
    }

    public void setTempoPorRodadaSegundos(int tempoPorRodadaSegundos) {
        this.tempoPorRodadaSegundos = tempoPorRodadaSegundos;
    }
}

package modelo;

import modelo.enumeradas.Objetivo;

import java.util.ArrayList;
import java.util.List;

// Um treino é um conjunto de exercicios feitos numa sessão (ex: "Peito e triceps" na segunda).
// Composição: o treino tem a lista de exercicios dele.
public class Treino implements Monitoravel {
    private int id;
    private String nome;
    private Objetivo objetivo;
    private String diaDaSemana;
    private List<Exercicio> exercicios = new ArrayList<>();

    public Treino(int id, String nome, Objetivo objetivo, String diaDaSemana) {
        this.id = id;
        this.nome = nome;
        this.objetivo = objetivo;
        this.diaDaSemana = diaDaSemana;
    }

    // soma as calorias de todos os exercicios do treino.
    // aqui é o polimorfismo mais claro: o treino nem sabe se é força, cardio ou funcional,
    // ele só chama estimarCaloriasGastas() e cada um responde do seu jeito
    @Override
    public double estimarCaloriasGastas() {
        double total = 0;
        for (Exercicio exercicio : exercicios) {
            total += exercicio.estimarCaloriasGastas();
        }
        return total;
    }

    public int getId() {
        return id;
    }

    public void setId(int id) {
        this.id = id;
    }

    public String getNome() {
        return nome;
    }

    public void setNome(String nome) {
        this.nome = nome;
    }

    public Objetivo getObjetivo() {
        return objetivo;
    }

    public void setObjetivo(Objetivo objetivo) {
        this.objetivo = objetivo;
    }

    public String getDiaDaSemana() {
        return diaDaSemana;
    }

    public void setDiaDaSemana(String diaDaSemana) {
        this.diaDaSemana = diaDaSemana;
    }

    public List<Exercicio> getExercicios() {
        return exercicios;
    }

    // cabeçalho da tabela de treinos
    public static String cabecalhoTabela() {
        return String.format("%-4s | %-20s | %-15s | %-8s | %-10s | %s",
                "ID", "Nome", "Objetivo", "Dia", "Exercicios", "Calorias (kcal)");
    }

    // linha da tabela com os dados do treino
    @Override
    public String toString() {
        return String.format("%-4d | %-20s | %-15s | %-8s | %-10d | %.1f",
                id, nome, objetivo, diaDaSemana, exercicios.size(), estimarCaloriasGastas());
    }
}

package modelo;

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

// Guarda que um aluno fez um treino em tal dia. É o historico,
// e é daqui que saem as estatisticas de frequencia e calorias.
public class RegistroDeTreino {
    // formato da data na tela, tipo 05/11/2026
    private static final DateTimeFormatter FORMATO_DATA = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    private int id;
    private Aluno aluno;
    private Treino treino;
    private LocalDate data;
    private double caloriasEstimadas;

    // as calorias já chegam calculadas. A gente guarda o valor aqui pra que,
    // se o treino mudar depois, o historico antigo não mude junto
    public RegistroDeTreino(int id, Aluno aluno, Treino treino, LocalDate data, double caloriasEstimadas) {
        this.id = id;
        this.aluno = aluno;
        this.treino = treino;
        this.data = data;
        this.caloriasEstimadas = caloriasEstimadas;
    }

    public int getId() {
        return id;
    }

    public void setId(int id) {
        this.id = id;
    }

    public Aluno getAluno() {
        return aluno;
    }

    public void setAluno(Aluno aluno) {
        this.aluno = aluno;
    }

    public Treino getTreino() {
        return treino;
    }

    public void setTreino(Treino treino) {
        this.treino = treino;
    }

    public LocalDate getData() {
        return data;
    }

    public void setData(LocalDate data) {
        this.data = data;
    }

    public double getCaloriasEstimadas() {
        return caloriasEstimadas;
    }

    public void setCaloriasEstimadas(double caloriasEstimadas) {
        this.caloriasEstimadas = caloriasEstimadas;
    }

    // cabeçalho da tabela de registros (fica junto do toString pras colunas alinharem)
    public static String cabecalhoTabela() {
        return String.format("%-4s | %-20s | %-20s | %-10s | %s",
                "ID", "Aluno", "Treino", "Data", "Calorias (kcal)");
    }

    // o registro como uma linha da tabela
    @Override
    public String toString() {
        return String.format("%-4d | %-20s | %-20s | %-10s | %.1f",
                id, aluno.getNome(), treino.getNome(), data.format(FORMATO_DATA), caloriasEstimadas);
    }
}

package repositorio;

import java.util.List;

// Interface generica com o CRUD basico. O T é o tipo guardado (Aluno, Treino, Exercicio, RegistroDeTreino).
// Os serviços só conhecem essa interface, não a implementação. Quem escolhe a implementação
// é o Programa, que cria o repositorio e passa pro serviço pelo construtor (inversão de controle / injeção de dependencia).
// Se um dia quiser salvar em arquivo ou banco, é só criar outra classe que implementa isso.
public interface RepositorioDeDados<T> {

    // guarda um objeto novo
    void inserir(T objeto);

    // troca o objeto que tem esse id pelo atualizado. false se o id não existir
    boolean atualizar(int id, T objetoAtualizado);

    // tira o objeto com esse id. false se não achou
    boolean remover(int id);

    // acha pelo id, ou null se não tiver
    T buscarPorId(int id);

    // devolve todo mundo que ta guardado
    List<T> listarTodos();
}

package repositorio;

import java.util.ArrayList;
import java.util.List;
import java.util.function.Function;

// Implementação do repositorio que guarda tudo na memoria, num ArrayList.
// Como a classe é generica ela não sabe sozinha pegar o id do objeto,
// entao recebe no construtor uma função que faz isso. Ex: new RepositorioEmLista<>(Aluno::getId)
public class RepositorioEmLista<T> implements RepositorioDeDados<T> {
    private ArrayList<T> dados = new ArrayList<>();
    private Function<T, Integer> obterId;

    public RepositorioEmLista(Function<T, Integer> obterId) {
        this.obterId = obterId;
    }

    // coloca no final da lista
    @Override
    public void inserir(T objeto) {
        dados.add(objeto);
    }

    // acha a posição do id e põe o objeto novo no lugar do antigo
    @Override
    public boolean atualizar(int id, T objetoAtualizado) {
        int posicao = buscarPosicao(id);
        if (posicao == -1) {
            return false;
        }
        dados.set(posicao, objetoAtualizado);
        return true;
    }

    // acha a posição e remove da lista
    @Override
    public boolean remover(int id) {
        int posicao = buscarPosicao(id);
        if (posicao == -1) {
            return false;
        }
        dados.remove(posicao);
        return true;
    }

    @Override
    public T buscarPorId(int id) {
        int posicao = buscarPosicao(id);
        if (posicao == -1) {
            return null;
        }
        return dados.get(posicao);
    }

    // devolve uma copia da lista, assim quem chamou pode ordenar à vontade sem bagunçar a original
    @Override
    public List<T> listarTodos() {
        return new ArrayList<>(dados);
    }

    // procura em que posição da lista ta o objeto com esse id (-1 se não tiver).
    // é usado no atualizar, remover e buscarPorId, pra não repetir o mesmo for 3 vezes
    private int buscarPosicao(int id) {
        for (int i = 0; i < dados.size(); i++) {
            Integer idDoObjeto = obterId.apply(dados.get(i)); // usa a função do construtor pra pegar o id
            if (idDoObjeto != null && idDoObjeto == id) {
                return i;
            }
        }
        return -1;
    }
}

package servico;

import modelo.Aluno;
import modelo.Treino;
import repositorio.RepositorioDeDados;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

// Regras de negocio dos alunos: cadastrar, alterar, remover, buscar e ordenar.
// O repositorio vem de fora pelo construtor (injeção de dependencia), o serviço não cria ele.
public class ServicoDeAlunos {
    private RepositorioDeDados<Aluno> repositorio;

    public ServicoDeAlunos(RepositorioDeDados<Aluno> repositorio) {
        this.repositorio = repositorio;
    }

    // dá um id novo pro aluno e guarda
    public Aluno cadastrar(Aluno aluno) {
        aluno.setId(gerarProximoId());
        repositorio.inserir(aluno);
        return aluno;
    }

    public boolean alterar(Aluno alunoAtualizado) {
        return repositorio.atualizar(alunoAtualizado.getId(), alunoAtualizado);
    }

    public boolean remover(int id) {
        return repositorio.remover(id);
    }

    public Aluno buscarPorId(int id) {
        return repositorio.buscarPorId(id);
    }

    public List<Aluno> listarTodos() {
        return repositorio.listarTodos();
    }

    // busca por atributo: alunos que tem esse pedaço no nome (sem ligar pra maiuscula/minuscula)
    public List<Aluno> buscarPorNome(String nome) {
        List<Aluno> resultado = new ArrayList<>();
        for (Aluno aluno : repositorio.listarTodos()) {
            if (aluno.getNome().toLowerCase().contains(nome.toLowerCase())) {
                resultado.add(aluno);
            }
        }
        return resultado;
    }

    // ordem alfabetica
    public List<Aluno> listarOrdenadoPorNome() {
        List<Aluno> lista = repositorio.listarTodos();
        lista.sort(Comparator.comparing(Aluno::getNome));
        return lista;
    }

    // do mais novo pro mais velho
    public List<Aluno> listarOrdenadoPorIdade() {
        List<Aluno> lista = repositorio.listarTodos();
        lista.sort(Comparator.comparingInt(Aluno::getIdade));
        return lista;
    }

    // quando um treino é excluido, ele tem que sair da lista de todos os alunos que tinham ele
    public void removerTreinoDosAlunos(Treino treino) {
        for (Aluno aluno : repositorio.listarTodos()) {
            aluno.getTreinos().remove(treino);
        }
    }

    // pega o maior id que já existe e soma 1 (se não tiver ninguem, começa no 1)
    private int gerarProximoId() {
        int maiorId = 0;
        for (Aluno aluno : repositorio.listarTodos()) {
            if (aluno.getId() > maiorId) {
                maiorId = aluno.getId();
            }
        }
        return maiorId + 1;
    }
}

    private RepositorioDeDados<Treino> repositorioTreinos;
    private RepositorioDeDados<Exercicio> repositorioExercicios;

    public ServicoDeTreinos(RepositorioDeDados<Treino> repositorioTreinos, RepositorioDeDados<Exercicio> repositorioExercicios) {
        this.repositorioTreinos = repositorioTreinos;
        this.repositorioExercicios = repositorioExercicios;
    }

    public Treino cadastrar(Treino treino) {
        treino.setId(gerarProximoId());
        repositorioTreinos.inserir(treino);
        return treino;
    }

    // coloca o exercicio no treino, mas só se ele estiver cadastrado. Devolve false se não achou
    public boolean adicionarExercicio(Treino treino, int idExercicio) {
        Exercicio exercicio = repositorioExercicios.buscarPorId(idExercicio);
        if (treino == null || exercicio == null) {
            return false;
        }
        treino.getExercicios().add(exercicio);
        return true;
    }

    // quando um exercicio é excluido, tira ele de todos os treinos que usavam
    public void removerExercicioDosTreinos(Exercicio exercicio) {
        for (Treino treino : repositorioTreinos.listarTodos()) {
            treino.getExercicios().remove(exercicio);
        }
    }

    public boolean alterar(Treino treinoAtualizado) {
        return repositorioTreinos.atualizar(treinoAtualizado.getId(), treinoAtualizado);
    }

    public boolean remover(int id) {
        return repositorioTreinos.remover(id);
    }

    public Treino buscarPorId(int id) {
        return repositorioTreinos.buscarPorId(id);
    }

    public List<Treino> listarTodos() {
        return repositorioTreinos.listarTodos();
    }

    // busca por atributo: todos os treinos com aquele objetivo
    public List<Treino> buscarPorObjetivo(Objetivo objetivo) {
        List<Treino> resultado = new ArrayList<>();
        for (Treino treino : repositorioTreinos.listarTodos()) {
            if (treino.getObjetivo() == objetivo) {
                resultado.add(treino);
            }
        }
        return resultado;
    }

    public List<Treino> listarOrdenadoPorNome() {
        List<Treino> lista = repositorioTreinos.listarTodos();
        lista.sort(Comparator.comparing(Treino::getNome));
        return lista;
    }

    // ordena na ordem que os objetivos aparecem no enum (hipertrofia, emagrecimento, condicionamento)
    public List<Treino> listarOrdenadoPorObjetivo() {
        List<Treino> lista = repositorioTreinos.listarTodos();
        lista.sort(Comparator.comparing(Treino::getObjetivo));
        return lista;
    }

    private int gerarProximoId() {
        int maiorId = 0;
        for (Treino treino : repositorioTreinos.listarTodos()) {
            if (treino.getId() > maiorId) {
                maiorId = treino.getId();
            }
        }
        return maiorId + 1;
    }
}

package servico;

import modelo.Aluno;
import modelo.RegistroDeTreino;
import modelo.Treino;
import repositorio.RepositorioDeDados;

import java.time.LocalDate;
import java.time.YearMonth;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

// Regras dos treinos realizados: registrar, alterar, remover, listar
// e as estatisticas (frequencia no mes, total de calorias, alunos parados).
// Igual aos outros, recebe o repositorio pelo construtor.
public class ServicoDeRegistros {
    private RepositorioDeDados<RegistroDeTreino> repositorio;

    public ServicoDeRegistros(RepositorioDeDados<RegistroDeTreino> repositorio) {
        this.repositorio = repositorio;
    }

    // registra que o aluno fez o treino nessa data.
    // regra: o treino tem que ser um dos treinos do aluno, senão devolve null.
    // as calorias são calculadas na hora com treino.estimarCaloriasGastas() (polimorfismo)
    public RegistroDeTreino registrarTreino(Aluno aluno, Treino treino, LocalDate data) {
        if (aluno == null || treino == null || !aluno.getTreinos().contains(treino)) {
            return null;
        }
        double calorias = treino.estimarCaloriasGastas();
        RegistroDeTreino registro = new RegistroDeTreino(gerarProximoId(), aluno, treino, data, calorias);
        repositorio.inserir(registro);
        return registro;
    }

    public boolean alterar(RegistroDeTreino registroAtualizado) {
        return repositorio.atualizar(registroAtualizado.getId(), registroAtualizado);
    }

    public boolean remover(int id) {
        return repositorio.remover(id);
    }

    // apaga o historico inteiro do aluno. Chamado quando o aluno é excluido,
    // pra não ficar registro de um aluno que nem existe mais
    public void removerRegistrosDoAluno(Aluno aluno) {
        for (RegistroDeTreino registro : listarPorAluno(aluno)) {
            repositorio.remover(registro.getId());
        }
    }

    public RegistroDeTreino buscarPorId(int id) {
        return repositorio.buscarPorId(id);
    }

    public List<RegistroDeTreino> listarTodos() {
        return repositorio.listarTodos();
    }

    // busca por atributo: todos os registros de um aluno
    public List<RegistroDeTreino> listarPorAluno(Aluno aluno) {
        List<RegistroDeTreino> resultado = new ArrayList<>();
        for (RegistroDeTreino registro : repositorio.listarTodos()) {
            if (registro.getAluno().equals(aluno)) {
                resultado.add(registro);
            }
        }
        return resultado;
    }

    // do mais antigo pro mais recente
    public List<RegistroDeTreino> listarOrdenadoPorData() {
        List<RegistroDeTreino> lista = repositorio.listarTodos();
        lista.sort(Comparator.comparing(RegistroDeTreino::getData));
        return lista;
    }

    // de quem gastou mais caloria pra quem gastou menos
    public List<RegistroDeTreino> listarOrdenadoPorCalorias() {
        List<RegistroDeTreino> lista = repositorio.listarTodos();
        lista.sort(Comparator.comparingDouble(RegistroDeTreino::getCaloriasEstimadas).reversed());
        return lista;
    }

    // quantas vezes o aluno treinou num mes especifico
    public int contarFrequenciaNoMes(Aluno aluno, YearMonth mes) {
        int contador = 0;
        for (RegistroDeTreino registro : listarPorAluno(aluno)) {
            if (YearMonth.from(registro.getData()).equals(mes)) { // YearMonth.from pega só mes/ano da data
                contador++;
            }
        }
        return contador;
    }

    // frequencia no mes atual, é a coluna "Frequencia (mes)" da tabela de alunos
    public int contarFrequenciaNoMesAtual(Aluno aluno) {
        return contarFrequenciaNoMes(aluno, YearMonth.now());
    }

    // soma das calorias de todos os treinos que o aluno já fez
    public double totalDeCaloriasDoAluno(Aluno aluno) {
        double total = 0;
        for (RegistroDeTreino registro : listarPorAluno(aluno)) {
            total += registro.getCaloriasEstimadas();
        }
        return total;
    }

    // dia do ultimo treino do aluno (null se nunca treinou)
    public LocalDate dataDoUltimoTreino(Aluno aluno) {
        LocalDate ultima = null;
        for (RegistroDeTreino registro : listarPorAluno(aluno)) {
            if (ultima == null || registro.getData().isAfter(ultima)) {
                ultima = registro.getData();
            }
        }
        return ultima;
    }

    // alerta de inatividade (diferencial): alunos que não treinam faz X dias ou que nunca treinaram
    public List<Aluno> listarAlunosInativos(List<Aluno> alunos, int dias) {
        LocalDate limite = LocalDate.now().minusDays(dias);
        List<Aluno> inativos = new ArrayList<>();
        for (Aluno aluno : alunos) {
            LocalDate ultimo = dataDoUltimoTreino(aluno);
            if (ultimo == null || ultimo.isBefore(limite)) {
                inativos.add(aluno);
            }
        }
        return inativos;
    }

    // mostra os registros em forma de tabela
    public void imprimirTabela(List<RegistroDeTreino> registros) {
        if (registros.isEmpty()) {
            System.out.println("Nenhum registro encontrado.");
            return;
        }
        System.out.println(RegistroDeTreino.cabecalhoTabela());
        for (RegistroDeTreino registro : registros) {
            System.out.println(registro); // chama o toString do registro
        }
    }

    private int gerarProximoId() {
        int maiorId = 0;
        for (RegistroDeTreino registro : repositorio.listarTodos()) {
            if (registro.getId() > maiorId) {
                maiorId = registro.getId();
            }
        }
        return maiorId + 1;
    }
}

    // pergunta quais treinos o aluno vai fazer. Enter em branco deixa como ta

        imprimirTreinos(servicoDeTreinos.listarTodos());
        List<Integer> ids = lerIds("IDs dos treinos separados por virgula (enter = pular)");
        if (ids == null) {
            return;

        aluno.getTreinos().clear();
        for (int id : ids) {
            Treino treino = servicoDeTreinos.buscarPorId(id);
            if (treino == null) {
                System.out.println("Treino " + id + " não existe, ignorei.");
            } else if (!aluno.getTreinos().contains(treino)) { // evita colocar o mesmo treino 2 vezes
                aluno.getTreinos().add(treino);
            }
        }

    private static void ordenarAlunos() {
        System.out.println("1. Por nome");
        System.out.println("2. Por idade");

            lista.sort(Comparator.comparingInt((Aluno a) -> servicoDeRegistros.contarFrequenciaNoMesAtual(a)).reversed());

    private static void alterarTreino() {
        Treino treino = servicoDeTreinos.buscarPorId(lerInteiro("ID do treino", 1, Integer.MAX_VALUE, null));

    private static void excluirTreino() {
        Treino treino = servicoDeTreinos.buscarPorId(lerInteiro("ID do treino", 1, Integer.MAX_VALUE, null));
        if (treino == null) {
            System.out.println("Treino não encontrado.");
            return;
        }
        if (confirmar("Excluir " + treino.getNome() + "? Ele sai da lista dos alunos.")) {
            // os registros antigos continuam, porque são historico do que já foi feito
            servicoDeAlunos.removerTreinoDosAlunos(treino);
            servicoDeTreinos.remover(treino.getId());
            System.out.println("Treino excluido.");
        }
    }

    // monta a lista de exercicios do treino a partir dos ids digitados.
    // se "obrigatorio" for true, fica pedindo até ter pelo menos um exercicio valido (validação de entrada)
    private static void escolherExerciciosDoTreino(Treino treino, boolean obrigatorio) {
        System.out.println("Exercicios disponiveis:");
        imprimirExercicios(servicoDeExercicios.listarTodos());
        while (true) {
            List<Integer> ids = lerIds("IDs dos exercicios separados por virgula" + (obrigatorio ? "" : " (enter = manter)"));
            if (ids == null) {
                if (!obrigatorio) {
                    return;
                }
                System.out.println("O treino precisa ter pelo menos um exercicio.");
                continue;
            }
            // guarda os antigos, caso nenhum id digitado preste
            List<Exercicio> antigos = new ArrayList<>(treino.getExercicios());
            treino.getExercicios().clear();
            for (int id : ids) {
                if (!servicoDeTreinos.adicionarExercicio(treino, id)) {
                    System.out.println("Exercicio " + id + " não existe, ignorei.");
                }
            }
            if (!treino.getExercicios().isEmpty()) {
                return;
            }
            treino.getExercicios().addAll(antigos);
            System.out.println("Nenhum id valido.");
            if (!obrigatorio) {
                return;
            }
        }
    }

            System.out.println(treino);
        }
    }

    private static void menuRegistros() {
        int opcao;
        do {
            System.out.println();

            System.out.println("1. Registrar treino realizado");
            System.out.println("2. Alterar data de um registro");
            System.out.println("3. Excluir registro");
            System.out.println("4. Listar");
            System.out.println("5. Buscar por aluno");
            System.out.println("6. Ordenar listagem");
            System.out.println("7. Alunos sem treinar (alerta de inatividade)");
            System.out.println("0. Voltar");
            opcao = lerInteiro("Opção", 0, 7, null);

            switch (opcao) {
                case 1 -> registrarTreino();
                case 2 -> alterarRegistro();
                case 3 -> excluirRegistro();
                case 4 -> servicoDeRegistros.imprimirTabela(servicoDeRegistros.listarTodos());
                case 5 -> buscarRegistrosDoAluno();
                case 6 -> ordenarRegistros();
                case 7 -> mostrarAlunosInativos();
            }
        } while (opcao != 0);
    }

    // escolhe o aluno, depois um dos treinos DELE, e a data. As calorias o serviço calcula sozinho
    private static void registrarTreino() {
        imprimirAlunos(servicoDeAlunos.listarTodos());
        Aluno aluno = servicoDeAlunos.buscarPorId(lerInteiro("ID do aluno", 1, Integer.MAX_VALUE, null));
        if (aluno == null) {
            System.out.println("Aluno não encontrado.");
            return;
        }
        if (aluno.getTreinos().isEmpty()) {
            System.out.println("Esse aluno ainda não tem treino. Coloque um treino nele no menu de alunos.");
            return;
        }
        imprimirTreinos(aluno.getTreinos());
        Treino treino = servicoDeTreinos.buscarPorId(lerInteiro("ID do treino", 1, Integer.MAX_VALUE, null));
        LocalDate data = lerData("Data (dd/mm/aaaa)", LocalDate.now()); // enter = hoje

        if (registro == null) {
            System.out.println("Esse treino não é do aluno.");
        } else {
            System.out.println(String.format("Treino registrado! Calorias estimadas: %.1f kcal", registro.getCaloriasEstimadas()));
        }
    }

    // monta o texto da pergunta, tipo "Nome [Ana Souza]: "
    private static String perguntar(String mensagem, Object valorAtual) {
        System.out.print(mensagem + (valorAtual == null ? "" : " [" + valorAtual + "]") + ": ");
        return entrada.nextLine().trim();
    }

    private static String lerTexto(String mensagem, String valorAtual) {
        while (true) {
            String texto = perguntar(mensagem, valorAtual);
            if (!texto.isEmpty()) {
                return texto;
            }
            if (valorAtual != null) {
                return valorAtual;
            }
            System.out.println("Não pode ficar em branco.");
        }
    }

    // lê um numero inteiro entre minimo e maximo. Se digitar letra ou fora da faixa, pergunta de novo
    private static int lerInteiro(String mensagem, int minimo, int maximo, Integer valorAtual) {
        while (true) {
            String texto = perguntar(mensagem, valorAtual);
            if (texto.isEmpty() && valorAtual != null) {
                return valorAtual;
            }
            try {
                int numero = Integer.parseInt(texto);
                if (numero >= minimo && numero <= maximo) {
                    return numero;
                }
                System.out.println("Digite um numero entre " + minimo + " e " + maximo + ".");
            } catch (NumberFormatException e) {
                System.out.println("Isso não é um numero valido.");
            }
        }
    }

    // igual ao de cima mas pra numero com virgula. Aceita tanto 62,5 quanto 62.5
    private static double lerDouble(String mensagem, double minimo, Double valorAtual) {
        while (true) {
            String texto = perguntar(mensagem, valorAtual);
            if (texto.isEmpty() && valorAtual != null) {
                return valorAtual;
            }
            try {
                double numero = Double.parseDouble(texto.replace(',', '.'));
                if (numero >= minimo) {
                    return numero;
                }
                System.out.println("O valor minimo é " + minimo + ".");
            } catch (NumberFormatException e) {
                System.out.println("Isso não é um numero valido.");
            }
        }
    }

    // lê uma data no formato dd/mm/aaaa. Não deixa colocar data no futuro
    private static LocalDate lerData(String mensagem, LocalDate valorAtual) {
        while (true) {
            String texto = perguntar(mensagem, valorAtual == null ? null : valorAtual.format(FORMATO_DATA));
            if (texto.isEmpty() && valorAtual != null) {
                return valorAtual;
            }
            try {
                LocalDate data = LocalDate.parse(texto, FORMATO_DATA);
                if (!data.isAfter(LocalDate.now())) {
                    return data;
                }
                System.out.println("A data não pode ser no futuro.");
            } catch (DateTimeParseException e) {
                System.out.println("Data invalida, use dd/mm/aaaa.");
            }
        }
    }

    // mostra as opções do enum numeradas e devolve a escolhida
    private static Objetivo lerObjetivo(Objetivo valorAtual) {
        Objetivo[] opcoes = Objetivo.values();
        for (int i = 0; i < opcoes.length; i++) {
            System.out.println((i + 1) + ". " + opcoes[i]);
        }
        int escolha = lerInteiro("Objetivo", 1, opcoes.length, valorAtual == null ? null : valorAtual.ordinal() + 1);
        return opcoes[escolha - 1];
    }

    private static Intensidade lerIntensidade(Intensidade valorAtual) {
        Intensidade[] opcoes = Intensidade.values();
        for (int i = 0; i < opcoes.length; i++) {
            System.out.println((i + 1) + ". " + opcoes[i]);
        }
        int escolha = lerInteiro("Intensidade", 1, opcoes.length, valorAtual == null ? null : valorAtual.ordinal() + 1);
        return opcoes[escolha - 1];
    }

    // lê uma lista de ids tipo "1, 4, 7". Devolve null se a pessoa só apertou enter
    private static List<Integer> lerIds(String mensagem) {
        while (true) {
            String texto = perguntar(mensagem, null);
            if (texto.isEmpty()) {
                return null;
            }
            try {
                List<Integer> ids = new ArrayList<>();
                for (String parte : texto.split(",")) {
                    ids.add(Integer.parseInt(parte.trim()));
                }
                return ids;
            } catch (NumberFormatException e) {
                System.out.println("Use só numeros separados por virgula, tipo 1, 3, 5");
            }
        }
    }

    // pergunta sim ou não antes de apagar alguma coisa
    private static boolean confirmar(String mensagem) {
        return perguntar(mensagem + " (s/n)", null).equalsIgnoreCase("s");
    }
